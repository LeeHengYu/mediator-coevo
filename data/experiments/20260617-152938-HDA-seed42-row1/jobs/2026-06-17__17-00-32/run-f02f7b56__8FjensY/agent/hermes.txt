+    date_pattern = r'(\d{4}-\d{2})(?:\s*~\s*(\d{4}-\d{2}))?'
+    match = re.search(date_pattern, phase_value)
+    
+    if match:
+        start_date_str = match.group(1)
+        end_date_str = match.group(2) if match.group(2) else start_date_str
+        
+        try:
+            start = datetime.strptime(start_date_str, '%Y-%m')
+            end = datetime.strptime(end_date_str, '%Y-%m')
+            
+            # Calculate months difference
+            months = (end.year - start.year) * 12 + (end.month - start.month) + 1

+            pass
