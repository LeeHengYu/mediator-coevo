+ts = wb['Task']

+# Data sheet layout observed in inspection:
+# Header row: 4
+# Series code column: D (rows 21:38)
+# Year headers: H4:L4
+# Numeric data range: H21:L38
+DATA_RANGE = 'Data!$H$21:$L$38'
+SERIES_RANGE = 'Data!$D$21:$D$38'
+YEAR_RANGE = 'Data!$H$4:$L$4'
+
+# Source data lookup formulas for admissions, discharges, and effective bed capacity.
+for row_start, row_end in [(12, 17), (19, 24), (26, 31)]:
+    for r in range(row_start, row_end + 1):
+        for c in range(8, 13):  # H:L
+            col = get_column_letter(c)
+            ts.cell(r, c).value = f'=INDEX({DATA_RANGE}, MATCH(D{r}, {SERIES_RANGE}, 0), MATCH({col}10, {YEAR_RANGE}, 0))'
+
+# Net Patient Flow: (Admissions - Discharges) / Bed Capacity * 100
+for idx, r in enumerate(range(35, 41)):
+    admit_row = 12 + idx
+    discharge_row = 19 + idx
+    bed_row = 26 + idx
+    for c in range(8, 13):
+        col = get_column_letter(c)
+        ts.cell(r, c).value = f'=({col}{admit_row}-{col}{discharge_row})/{col}{bed_row}*100'

+# Statistics labels are in column E in the workbook (C is blank for rows 42:47).

+    'simple mean': 'AVERAGE',
