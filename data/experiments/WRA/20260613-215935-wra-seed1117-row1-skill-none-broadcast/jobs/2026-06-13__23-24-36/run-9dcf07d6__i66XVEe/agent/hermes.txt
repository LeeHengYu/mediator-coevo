+def percentile_linear(values, percent):
+    """Match numpy.percentile default / Excel PERCENTILE.INC interpolation."""
+    xs = sorted(values)
+    if len(xs) == 1:
+        return xs[0]
+    pos = (len(xs) - 1) * (percent / 100.0)
+    lo = int(pos)
+    hi = min(lo + 1, len(xs) - 1)
+    frac = pos - lo
+    return xs[lo] * (1 - frac) + xs[hi] * frac
