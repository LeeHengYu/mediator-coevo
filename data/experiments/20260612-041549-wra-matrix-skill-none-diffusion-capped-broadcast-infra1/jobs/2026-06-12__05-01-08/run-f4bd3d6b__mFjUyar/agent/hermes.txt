+def percentile_inc(values, k):
+    """Excel PERCENTILE/PERCENTILE.INC semantics for k in [0, 1]."""
+    vals = sorted(float(v) for v in values)
+    n = len(vals)
+    if n == 0:
+        raise ValueError('empty values')
+    if n == 1:
+        return vals[0]
+    pos = 1 + (n - 1) * k
+    lo = int(pos)
+    frac = pos - lo
+    if frac == 0:
+        return vals[lo - 1]
+    return vals[lo - 1] + frac * (vals[lo] - vals[lo - 1])
