+    """
+    Compute Korean full-year age (만 나이) as of visit date.
+    age = visit_year - birth_year, adjusted down by 1 if visit is before birthday.
+    """
+    birth_date = datetime.strptime(birth_date_str, '%Y-%m-%d')
+    visit_date = datetime.strptime(visit_date_str, '%Y-%m-%d')
+    
+    age = visit_date.year - birth_date.year
+    if (visit_date.month, visit_date.day) < (birth_date.month, birth_date.day):
+        age -= 1
+    
+    return age

+def normalize_phone(phone_str):

+    Normalize phone number to 000-0000-0000 format.
+    Strip all non-digit characters first.
+    For 11 digits (like 01012345678): 010-1234-5678
