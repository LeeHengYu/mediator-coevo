+    """
+    Calculate inclusive month span from date range.
+    Format: 'YYYY-MM ~ YYYY-MM' or 'YYYY-MM'
+    Returns: number of months (inclusive)
+    """
+    # Match date range pattern
+    range_match = re.search(r'(\d{4})-(\d{2})\s*~\s*(\d{4})-(\d{2})', date_range_str)
+    if range_match:
+        start_year, start_month, end_year, end_month = map(int, range_match.groups())
+        months = (end_year - start_year) * 12 + (end_month - start_month) + 1
+        return months
+    
+    # Single month pattern
+    single_match = re.search(r'(\d{4})-(\d{2})', date_range_str)
+    if single_match:
+        return 1
+    
+    return None
