+    department = data['진료팀']

+    symptoms = data['증상요약']

+    # Compute Korean age: visit_year - birth_year, minus 1 if before birthday
+    birth_year, birth_month, birth_day = map(int, birth_date.split('-'))
+    visit_year, visit_month, visit_day = map(int, visit_date.split('-'))
+    
+    age = visit_year - birth_year
+    # Subtract 1 if visit date is before birthday in that year
+    if (visit_month, visit_day) < (birth_month, birth_day):
+        age -= 1
+    
+    age_note = f"({age}세)"
+    
+    # Normalize phone number: strip non-digits, format as 000-0000-0000
+    digits = re.sub(r'\D', '', phone_raw)
+    if len(digits) == 11:
